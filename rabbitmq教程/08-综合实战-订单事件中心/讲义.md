# 第 8 讲 · 综合实战:订单事件中心

前七讲的零件合成一套典型的电商消息架构。

## 0 架构

```
                                 ┌────────── shop.events(topic) ──────────┐
  order_service ── order.paid ──►│──┬──► shop.sms.queue ──► sms_service     │
  (下单/付款,广播事件)           │  │      (失败重试×2,再败死信)           │
                                 │  └──► shop.points.queue ──► points_service│
                                 └────────────────────────────────────────┘

  order_service ── 延迟炸弹 ──► shop.pay.wait(TTL 15 秒)
                                     │ 到期 expired 死信,路由键改写为 timeout
                                     ▼
                     shop.dlx(direct)──┬── timeout ──► shop.timeout.queue ──► timeout_service
                                       └── dead ─────► shop.dead.queue ────► watcher
```

拓扑配置:

| 对象 | 类型 | 关键参数 | 出处 |
|---|---|---|---|
| `shop.events` | topic 交换机 | durable | 第 3 讲 |
| `shop.dlx` | direct 交换机 | durable,两个出口 dead / timeout | 第 5 讲 |
| `shop.sms.queue` | 队列 | 绑 `order.paid`;DLX=shop.dlx,routing-key=dead | 第 5 讲 |
| `shop.points.queue` | 队列 | 绑 `order.paid` | 第 3 讲 |
| `shop.pay.wait` | 队列 | `x-message-ttl=15000`;DLX=shop.dlx,routing-key=timeout | 第 6 讲 |
| `shop.timeout.queue` | 队列 | 绑 dlx / `timeout` | 第 6 讲 |
| `shop.dead.queue` | 队列 | 绑 dlx / `dead` | 第 5 讲 |

三笔订单的剧本(写死在 `order_service.py`):

| 订单 | 剧本 | 预期结局 |
|---|---|---|
| A2001 | 5 秒后付款 | 短信发出、积分到账;15 秒到点时回查发现已付款,不取消 |
| A2002 | 不付款 | 15 秒后自动取消、释放库存 |
| A2003 | 8 秒后付款,短信注定失败 | 短信重试三次全败 → 死信 → watcher 告警;积分照常发放 |

## 1 文件清单

| 文件 | 职责 |
|---|---|
| `common.py` | 连接地址 + 全套拓扑的幂等声明,所有脚本共用 |

| `points_service.py` | 消费者:加积分;数据库版幂等 |
| `timeout_service.py` | 消费者:延迟到点回查订单,没付才取消 |
| `watcher.py` | 消费者:死信告警 |
| `reset.py` | 一键清场(仓库自带,不贴讲义) |

## 2 common.py —— 拓扑声明

**要点**:声明是幂等的,五六个进程同时启动、同时建同一套拓扑不会冲突,**没有启动顺序依赖**。真实项目里对应一个集中定义的 `topology.py`。

```python
"""第 8 讲公共零件:连接地址 + 全套拓扑的幂等声明。

每个服务脚本启动时都调一遍 declare_topology,谁先启动谁建。
"""

import os
from pathlib import Path

import aio_pika
from dotenv import load_dotenv

load_dotenv(Path(__file__).parents[2] / ".env")
RABBITMQ_URL = os.environ["RABBITMQ_URL"]

EXCHANGE = "shop.events"            # 业务事件交换机(topic)
DLX = "shop.dlx"                    # 死信交换机(direct),两个出口:dead / timeout
SMS_QUEUE = "shop.sms.queue"        # 短信队列,绑 order.paid;失败死信走 dead
POINTS_QUEUE = "shop.points.queue"  # 积分队列,绑 order.paid
PAY_WAIT = "shop.pay.wait"          # 付款等待队列:15 秒 TTL,到期死信走 timeout
DEAD_QUEUE = "shop.dead.queue"      # 死信池,人工排查入口
TIMEOUT_QUEUE = "shop.timeout.queue"  # 超时处理队列


async def declare_topology(channel: aio_pika.Channel) -> dict[str, object]:
    """声明本讲全部交换机、队列和绑定,返回各对象供调用方使用"""
    events = await channel.declare_exchange(EXCHANGE, aio_pika.ExchangeType.TOPIC, durable=True)
    dlx = await channel.declare_exchange(DLX, aio_pika.ExchangeType.DIRECT, durable=True)

    sms = await channel.declare_queue(
        SMS_QUEUE,
        durable=True,
        # 重试耗尽的死信走 dead 出口
        arguments={"x-dead-letter-exchange": DLX, "x-dead-letter-routing-key": "dead"},
    )
    await sms.bind(events, routing_key="order.paid")

    points = await channel.declare_queue(POINTS_QUEUE, durable=True)
    await points.bind(events, routing_key="order.paid")

    pay_wait = await channel.declare_queue(
        PAY_WAIT,
        durable=True,
        arguments={
            "x-message-ttl": 15_000,                 # 躺 15 秒
            "x-dead-letter-exchange": DLX,
            "x-dead-letter-routing-key": "timeout",  # 到期死信走 timeout 出口
        },
    )

    timeout_queue = await channel.declare_queue(TIMEOUT_QUEUE, durable=True)
    await timeout_queue.bind(dlx, routing_key="timeout")
    dead_queue = await channel.declare_queue(DEAD_QUEUE, durable=True)
    await dead_queue.bind(dlx, routing_key="dead")

    return {
        "events": events,
        "dlx": dlx,
        "sms": sms,
        "points": points,
        "pay_wait": pay_wait,
        "timeout": timeout_queue,
        "dead": dead_queue,
    }
```

**同一个 DLX 两个出口**靠 `x-dead-letter-routing-key` 分流,这也是死信交换机用 direct 而非 fanout 的原因。

## 3 db.py —— 假装是 MySQL

**要点**:`points_granted` 字段是积分服务的幂等标记,`sms_will_fail` 是演示剧本开关。

```python
"""极简订单表:一个 json 文件假装是数据库。

真实项目里这就是 MySQL 的 orders 表,每个函数换成一条 SQL。
"""

import json
from pathlib import Path

DB_FILE = Path(__file__).parent / "orders.json"


def load() -> dict:
    if not DB_FILE.exists():
        return {}
    return json.loads(DB_FILE.read_text(encoding="utf-8"))


def save(orders: dict) -> None:
    DB_FILE.write_text(json.dumps(orders, ensure_ascii=False, indent=2), encoding="utf-8")


def get(order_id: str) -> dict | None:
    return load().get(order_id)


def create(order_id: str, amount: int, sms_will_fail: bool = False) -> None:
    orders = load()
    orders[order_id] = {
        "order_id": order_id,
        "amount": amount,
        "status": "待付款",
        "points_granted": False,    # 积分服务的幂等标记
        "sms_will_fail": sms_will_fail,
    }
    save(orders)


def mark_paid(order_id: str) -> None:
    orders = load()
    orders[order_id]["status"] = "已付款"
    save(orders)


def grant_points(order_id: str) -> bool:
    """发放积分。已发过返回 False —— 积分服务靠它做幂等"""
    orders = load()
    if order_id not in orders or orders[order_id]["points_granted"]:
        return False
    orders[order_id]["points_granted"] = True
    save(orders)
    return True


def cancel(order_id: str, reason: str) -> None:
    orders = load()
    orders[order_id]["status"] = "已取消"
    orders[order_id]["cancel_reason"] = reason
    save(orders)
```

## 4 order_service.py —— 生产者

**三个动作**:

| 动作 | 目标 | 要点 |
|---|---|---|
| 广播 `order.created` | shop.events | 眼下没有队列绑它,消息发出即丢弃。留的扩展口:将来数据分析自己绑队列即可,上游一行不改 |
| 投延迟炸弹 | shop.pay.wait(直投) | 第 6 讲的延迟套路 |
| 付款 | 先改库,**再**广播 `order.paid` | 顺序反了,下游收到事件去查库还是"待付款" |

```python
"""第 8 讲·订单服务(生产者):下单、模拟付款,把事件发进 shop.events。

    uv run python "rabbitmq教程/08-综合实战-订单事件中心/order_service.py"
"""

import asyncio
import json
from datetime import datetime
from pathlib import Path

import aio_pika
import db
from common import PAY_WAIT, RABBITMQ_URL, declare_topology
from dotenv import load_dotenv

load_dotenv(Path(__file__).parents[2] / ".env")

# (订单号, 金额, 短信会不会失败, 用户会不会付款)
ORDERS = [
    ("A2001", 199, False, True),
    ("A2002", 59, False, False),
    ("A2003", 299, True, True),
]


def now() -> str:
    return f"[{datetime.now():%H:%M:%S}]"


def event_message(payload: dict, **kwargs) -> aio_pika.Message:
    """业务事件统一打包:json 内容 + 持久化"""
    return aio_pika.Message(
        json.dumps(payload, ensure_ascii=False).encode(),
        delivery_mode=aio_pika.DeliveryMode.PERSISTENT,
        **kwargs,
    )


async def pay(channel: aio_pika.Channel, events: aio_pika.Exchange, order_id: str) -> None:
    """付款:先改库,再广播事件,顺序别反"""
    db.mark_paid(order_id)
    await events.publish(
        event_message({"order_id": order_id, "event": "order.paid"}), routing_key="order.paid"
    )
    print(f"{now()} 用户付款:{order_id} → 广播 order.paid")


async def main() -> None:
    connection = await aio_pika.connect_robust(RABBITMQ_URL)
    async with connection:
        channel = await connection.channel()
        topo = await declare_topology(channel)
        events: aio_pika.Exchange = topo["events"]  # type: ignore[assignment]

        for order_id, amount, sms_will_fail, _will_pay in ORDERS:
            db.create(order_id, amount, sms_will_fail)
            # 事件一:order.created(暂无订阅方)
            await events.publish(
                event_message({"order_id": order_id, "event": "order.created"}),
                routing_key="order.created",
            )
            # 事件二:延迟炸弹,直投付款等待队列
            await channel.default_exchange.publish(
                event_message({"order_id": order_id, "event": "payment.deadline"}),
                routing_key=PAY_WAIT,
            )
            print(f"{now()} 下单:{order_id},{amount} 元,15 秒付款窗口开始")

        print(f"{now()} 模拟付款:5 秒后 A2001,8 秒后 A2003")
        await asyncio.sleep(5)
        await pay(channel, events, "A2001")
        await asyncio.sleep(3)
        await pay(channel, events, "A2003")

    print(f"{now()} 订单服务退出,消费者继续在各自终端运行")


asyncio.run(main())
```

## 5 sms_service.py —— 失败处理三板斧

| 手段 | 实现 | 说明 |
|---|---|---|
| 幂等挡板 | 内存集合 `sent` | **最弱的幂等**,进程重启即失效;生产换 Redis SET 或数据库唯一键 |
| 重试 | ack 旧消息,带 `retry` 计数的新消息直投本队列 | 消息进队列后**无法修改**,想带计数只能"发新的、确认旧的" |
| 放弃 | `nack(requeue=False)` | 队列的 DLX 接手,进死信池,watcher 告警 |

重试为什么直投 `SMS_QUEUE` 而不是发布到 `shop.events`?重试是短信服务自己的私事,广播出去积分服务会再加一次积分。

```python
"""第 8 讲·短信服务(消费者):听 order.paid,发支付成功短信。

  失败重试两次,第三次仍失败 → nack(requeue=False) → 死信池
  幂等:内存集合(最弱版本,和积分服务的持久版对照看)

    uv run python "rabbitmq教程/08-综合实战-订单事件中心/sms_service.py"
"""

import asyncio
import json
from datetime import datetime

import aio_pika
import db
from common import RABBITMQ_URL, SMS_QUEUE, declare_topology

sent: set[str] = set()   # 本进程已发过短信的订单,重启清零


def now() -> str:
    return f"[{datetime.now():%H:%M:%S}]"


async def main() -> None:
    connection = await aio_pika.connect_robust(RABBITMQ_URL)
    async with connection:
        channel = await connection.channel()
        await channel.set_qos(prefetch_count=1)
        topo = await declare_topology(channel)
        sms: aio_pika.Queue = topo["sms"]  # type: ignore[assignment]

        print(f"{now()} 短信服务已启动")
        async for message in sms.iterator():
            event = json.loads(message.body)
            order_id = event["order_id"]
            order = db.get(order_id)
            retry = message.headers.get("retry", 0) if message.headers else 0

            if order_id in sent:                      # 幂等挡板
                print(f"{now()} 短信:{order_id} 已发过,跳过")
                await message.ack()
                continue

            if order and order["sms_will_fail"]:
                if retry < 2:                         # 重试
                    print(f"{now()} 短信:{order_id} 失败(第 {retry + 1} 次),带计数重发")
                    await channel.default_exchange.publish(
                        aio_pika.Message(
                            message.body,
                            headers={"retry": retry + 1},
                            delivery_mode=aio_pika.DeliveryMode.PERSISTENT,
                        ),
                        routing_key=SMS_QUEUE,        # 直投本队列,不广播
                    )
                    await message.ack()               # 旧消息以"重发"的方式处理完毕
                else:                                 # 放弃
                    print(f"{now()} 短信:{order_id} 三次全败 → nack(requeue=False) → 死信")
                    await message.nack(requeue=False)
                continue

            amount = order["amount"] if order else "?"
            print(f"{now()} 短信:【云上小店】订单 {order_id} 已支付 {amount} 元")
            sent.add(order_id)
            await message.ack()


asyncio.run(main())
```

延伸:想改成"隔 5 秒再试",把直投目标换成一条 `x-message-ttl=5000`、死信再指回 `SMS_QUEUE` 的中转队列即可(第 6 讲的分档延迟)。配上 15s / 60s 两档就是常见的阶梯重试。

## 6 points_service.py —— 持久化幂等

**铁律:先落库,后 ack**。反过来的话,落库失败但消息已确认,这条消息凭空蒸发。

```python
"""第 8 讲·积分服务(消费者):听 order.paid,按金额发放积分。

幂等标记写在订单表里:进程随便重启、消息随便重投,积分只发一次。

    uv run python "rabbitmq教程/08-综合实战-订单事件中心/points_service.py"
"""

import asyncio
import json
from datetime import datetime

import aio_pika
import db
from common import RABBITMQ_URL, declare_topology


def now() -> str:
    return f"[{datetime.now():%H:%M:%S}]"


async def main() -> None:
    connection = await aio_pika.connect_robust(RABBITMQ_URL)
    async with connection:
        channel = await connection.channel()
        await channel.set_qos(prefetch_count=1)
        topo = await declare_topology(channel)
        points: aio_pika.Queue = topo["points"]  # type: ignore[assignment]

        print(f"{now()} 积分服务已启动")
        async for message in points.iterator():
            event = json.loads(message.body)
            order_id = event["order_id"]
            order = db.get(order_id)

            if db.grant_points(order_id):       # 先落库
                print(f"{now()} 积分:{order_id} 发放 {order['amount']} 分")
            else:
                print(f"{now()} 积分:{order_id} 已发过,跳过(表里标记拦下,重启也拦得住)")
            await message.ack()                 # 后 ack


asyncio.run(main())
```

两种幂等对比:

| | sms_service | points_service |
|---|---|---|
| 存储 | 进程内存 `set` | 订单表字段 |
| 进程重启后 | 失效,会重复发短信 | 仍然有效 |
| 生产可用 | 否 | 是(真实项目用 Redis SET / 数据库唯一键) |

## 7 timeout_service.py —— 到点回查

第 6 讲的规矩落地:到点只代表"该看一眼",三个分支。

```python
"""第 8 讲·超时服务(消费者):听 15 秒延迟到期的死信,回查订单再决定取消。

    uv run python "rabbitmq教程/08-综合实战-订单事件中心/timeout_service.py"
"""

import asyncio
import json
from datetime import datetime

import aio_pika
import db
from common import RABBITMQ_URL, declare_topology


def now() -> str:
    return f"[{datetime.now():%H:%M:%S}]"


async def main() -> None:
    connection = await aio_pika.connect_robust(RABBITMQ_URL)
    async with connection:
        channel = await connection.channel()
        await channel.set_qos(prefetch_count=1)
        topo = await declare_topology(channel)
        timeout_queue: aio_pika.Queue = topo["timeout"]  # type: ignore[assignment]

        print(f"{now()} 超时服务已启动")
        async for message in timeout_queue.iterator():
            event = json.loads(message.body)
            order_id = event["order_id"]
            order = db.get(order_id)
            status = order["status"] if order else "不存在"

            if status == "已付款":
                print(f"{now()} 超时:{order_id} 到点但已付款,不处理")
            elif status == "已取消":
                print(f"{now()} 超时:{order_id} 已取消过,跳过(幂等)")
            else:
                db.cancel(order_id, "超时未付款")
                print(f"{now()} 超时:{order_id} 15 秒未付款 → 取消,释放库存")
            await message.ack()


asyncio.run(main())
```

## 8 watcher.py —— 死信告警

第 5 讲"收容之后呢"的答案。生产里 `print` 换成钉钉 / 飞书机器人。

```python
"""第 8 讲·死信观察员:守着死信池,消息一到就告警。

    uv run python "rabbitmq教程/08-综合实战-订单事件中心/watcher.py"
"""

import asyncio
from datetime import datetime

import aio_pika
from common import RABBITMQ_URL, declare_topology


def now() -> str:
    return f"[{datetime.now():%H:%M:%S}]"


async def main() -> None:
    connection = await aio_pika.connect_robust(RABBITMQ_URL)
    async with connection:
        channel = await connection.channel()
        topo = await declare_topology(channel)
        dead: aio_pika.Queue = topo["dead"]  # type: ignore[assignment]

        print(f"{now()} 死信观察员已启动")
        async for message in dead.iterator():
            print(f"{now()} 死信到达,推送告警:")
            print(f"    消息体: {message.body.decode('utf-8')}")
            for death in message.headers.get("x-death", []):
                reason = death.get("reason")
                reason = reason.decode() if isinstance(reason, bytes) else reason
                origin = death.get("queue")
                origin = origin.decode() if isinstance(origin, bytes) else origin
                print(f"    档案: 死于 {reason},原队列 {origin}")
            await message.ack()


asyncio.run(main())
```
