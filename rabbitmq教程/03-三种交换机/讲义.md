# 第 3 讲 · 三种交换机

前两讲消息都直进队列,是因为一直在用那个"路由键写队列名"的默认交换机。真实的消息架构里,中间站着**交换机**:

```
                                    ch3.orders(topic 交换机)
   订单服务 ──publish──►  ( 消息 + 路由键 )──►  按绑定规则分发  ──┬──► 队列A  绑定键 order.created
                                                                 ├──► 队列B  绑定键 order.*
                                                                 └──► 队列C  绑定键 #
```

> **敲黑板:** 生产者只跟交换机打交道,消息进哪个队列,由交换机类型和绑定关系决定。绑定(binding)就是一条登记规则:"队列 A 想要路由键长这样的消息"。这一讲三种交换机各跑一个脚本,企业里 90% 的路由需求,就是这三种里的某一种。

三个脚本结构相同:声明交换机 → 声明队列并绑定 → 发消息 → 把每个队列收到的倒出来看 → 删掉演示资源。全文如下,三个都值得各敲一遍,敲到第三个,声明、绑定、收发这套动作就熟练了。

## 完整代码

`fanout.py`(全文),广播:

```python
"""第 3 讲·fanout 交换机:广播——绑了它的队列,每条都收一份副本。

uv run python "rabbitmq教程/03-三种交换机/fanout.py"
"""

import asyncio
import os
from pathlib import Path

import aio_pika
from dotenv import load_dotenv

load_dotenv(Path(__file__).parents[2] / ".env")
RABBITMQ_URL = os.environ["RABBITMQ_URL"]

# 三条日志就是全部的"业务数据",直接写在脚本里
LOGS = [
    "[WARN] 库存服务响应变慢",
    "[INFO] 用户 1001 登录成功",
    "[ERROR] 支付回调第三次重试",
]


async def drain(queue: aio_pika.Queue) -> list[str]:
    """把队列里现成的消息全部取出来(取走即确认,演示够用)"""
    messages = []
    while True:
        message = await queue.get(no_ack=True, fail=False, timeout=2)  # ⑤ 主动拉一条
        if message is None:
            return messages
        messages.append(message.body.decode("utf-8"))


async def main() -> None:
    connection = await aio_pika.connect(RABBITMQ_URL)
    async with connection:
        channel = await connection.channel()

        exchange = await channel.declare_exchange(
            "ch3.logs", aio_pika.ExchangeType.FANOUT, durable=True
        )

        # 两个订阅者:一个负责归档,一个负责实时上屏。绑定时 routing_key 随便,fanout 根本不看
        archive = await channel.declare_queue("ch3.log.archive", durable=True)  # ② 队列一:归档
        console = await channel.declare_queue("ch3.log.console", durable=True)
        await archive.purge()
        await console.purge()
        await archive.bind(exchange, routing_key="")  # ③ 绑定:登记到交换机名下
        await console.bind(exchange, routing_key="")

        for log in LOGS:
            await exchange.publish(
                aio_pika.Message(log.encode("utf-8")),
                routing_key="",  # fanout 下这个参数形同虚设
            )
            print(f"已广播: {log}")

        print("\n广播完毕,看看两个订阅者各自收到了什么:")
        for name, queue in [("归档库", archive), ("控制台", console)]:
            received = await drain(queue)
            print(f"\n[{name}] 收到 {len(received)} 条(和发出的一模一样):")
            for text in received:
                print(f"    {text}")

        # 演示结束,把本讲的队列和交换机删掉,保持 broker 干净
        await archive.delete()
        await console.delete()
        await exchange.delete()


asyncio.run(main())
```

`direct.py`(全文),定向:

```python
"""第 3 讲·direct 交换机:定向——路由键完全相等,才投给谁。

uv run python "rabbitmq教程/03-三种交换机/direct.py"
"""

import asyncio
import os
from pathlib import Path

import aio_pika
from dotenv import load_dotenv

load_dotenv(Path(__file__).parents[2] / ".env")
RABBITMQ_URL = os.environ["RABBITMQ_URL"]

# (路由键, 消息内容):一条消息该走哪个通道,由路由键说了算
NOTIFICATIONS = [
    ("sms", "验证码 8848,五分钟内有效"),
    ("email", "您本月的账单已出,附件为明细"),
    ("sms", "您的快递已到丰巢,取件码 3306"),
    ("push", "这条走 push 通道——但没人绑定 push,它会被直接丢弃"),
]


async def drain(queue: aio_pika.Queue) -> list[str]:
    messages = []
    while True:
        message = await queue.get(no_ack=True, fail=False, timeout=2)
        if message is None:
            return messages
        messages.append(message.body.decode("utf-8"))


async def main() -> None:
    connection = await aio_pika.connect(RABBITMQ_URL)
    async with connection:
        channel = await connection.channel()

        exchange = await channel.declare_exchange(
            "ch3.notify", aio_pika.ExchangeType.DIRECT, durable=True
        )

        # 短信队列只绑 "sms",邮件队列只绑 "email"。没有队列绑 "push"
        sms = await channel.declare_queue("ch3.sms", durable=True)
        email = await channel.declare_queue("ch3.email", durable=True)
        await sms.purge()
        await email.purge()
        await sms.bind(exchange, routing_key="sms")  # ② 各绑各的键:短信队列只认 sms
        await email.bind(exchange, routing_key="email")

        for routing_key, text in NOTIFICATIONS:
            await exchange.publish(aio_pika.Message(text.encode("utf-8")), routing_key=routing_key)
            print(f"已发送 → 路由键[{routing_key}]: {text}")

        print("\n投递完毕,看看谁收到了什么:")
        for name, queue in [("短信队列", sms), ("邮件队列", email)]:
            received = await drain(queue)
            print(f"\n[{name}] 收到 {len(received)} 条:")
            for text in received:
                print(f"    {text}")

        print("\n走 push 的那条,两边都没收到——路由不到任何队列的消息,direct 交换机直接丢弃")
        print("(怎么感知「路由失败」?第 4 讲的 mandatory 讲这个)")

        await sms.delete()
        await email.delete()
        await exchange.delete()


asyncio.run(main())
```

`topic.py`(全文),通配符:

```python
"""第 3 讲·topic 交换机:通配符——路由键按点分词,* 配一个词,# 配零个或多个词。

uv run python "rabbitmq教程/03-三种交换机/topic.py"
"""

import asyncio
import os
from pathlib import Path

import aio_pika
from dotenv import load_dotenv

load_dotenv(Path(__file__).parents[2] / ".env")
RABBITMQ_URL = os.environ["RABBITMQ_URL"]

# (路由键, 消息内容)
EVENTS = [
    ("order.created", "订单 A1001 创建"),
    ("order.paid", "订单 A1001 支付成功"),
    ("order.cancelled", "订单 B2002 取消"),
    ("refund.created", "退款单 R3003 创建"),
]


async def drain(queue: aio_pika.Queue) -> list[str]:
    messages = []
    while True:
        message = await queue.get(no_ack=True, fail=False, timeout=2)
        if message is None:
            return messages
        messages.append(message.body.decode("utf-8"))


async def main() -> None:
    connection = await aio_pika.connect(RABBITMQ_URL)
    async with connection:
        channel = await connection.channel()

        exchange = await channel.declare_exchange(
            "ch3.orders", aio_pika.ExchangeType.TOPIC, durable=True
        )

        # 三个订阅者,口味一个比一个广
        watch_created = await channel.declare_queue("ch3.watch.created", durable=True)  # 只盯新订单
        watch_orders = await channel.declare_queue(
            "ch3.watch.orders", durable=True
        )  # 订单的所有动作
        watch_all = await channel.declare_queue("ch3.watch.all", durable=True)  # 全站所有事件
        for q in (watch_created, watch_orders, watch_all):
            await q.purge()

        await watch_created.bind(exchange, routing_key="order.created")  # 不含通配符:精确匹配
        await watch_orders.bind(exchange, routing_key="order.*")  # order.一个词
        await watch_all.bind(exchange, routing_key="#")  # 井号:来者不拒

        for routing_key, text in EVENTS:
            await exchange.publish(aio_pika.Message(text.encode("utf-8")), routing_key=routing_key)
            print(f"已发送 → [{routing_key}]: {text}")

        print("\n投递完毕,三个订阅者各自收到了:")
        for name, queue in [
            ("只盯下单 order.created", watch_created),
            ("订单动作 order.*", watch_orders),
            ("全站事件 #", watch_all),
        ]:
            received = await drain(queue)
            print(f"\n[{name}] 收到 {len(received)} 条:")
            for text in received:
                print(f"    {text}")

        await watch_created.delete()
        await watch_orders.delete()
        await watch_all.delete()
        await exchange.delete()


asyncio.run(main())
```

## 逐段讲

### fanout——广播

看 fanout.py 的 ①②③④:交换机声明成 `ExchangeType.FANOUT`(①),两个队列(归档库、控制台)都绑上去(③),绑定时 `routing_key=""`,fanout 根本不看路由键——凡是绑到这个交换机上的队列,每条消息都收到一份完整副本(④ 的路由键纯属摆设)。

适用场景:一条"配置变更了"的消息,同时让十台机器刷新缓存;日志同时进归档库和实时大屏。`fanout.py` 跑出来两个队列各 3 条,一条不差。

### direct——定向

direct.py 的 ①②③:交换机声明成 `ExchangeType.DIRECT`(①),短信队列绑 `sms`、邮件队列绑 `email`(②),消息的路由键和绑定键完全相等,才投给它(③)。四条消息发出去:短信队列收到 2 条、邮件队列 1 条,而走 `push` 的那条两边都没收到,broker 找不到任何匹配的绑定,直接丢弃,连个招呼都不打。

> **敲黑板:** 路由不到任何队列的消息,默认直接丢弃,生产端毫无知觉。想感知"路由失败"得开 mandatory,第 4 讲的实验专门做它。

### topic——通配符,企业里的主力

topic.py 的 ②③:路由键按 `.` 分词,绑定键里可以有两个通配符:

- `*` 恰好匹配**一个**词:`order.*` 配得上 `order.created`、`order.paid`,配不上 `order` 也不配 `order.paid.timeout`;
- `#` 匹配**零个或多个**词:`order.#` 配 `order`、`order.paid`、`order.paid.timeout` 都行;单独一个 `#` 就是来者不拒。

三个订阅者把梯度演示得很清楚:绑 `order.created` 的只收 1 条,绑 `order.*` 的收 3 条(所有 order.单步),绑 `#` 的 4 条全收(连 refund.created 也进来了)。

顺带说一句:第 1 讲的默认交换机其实就是个 direct 交换机,只不过 broker 预先帮你把每个队列都用"自己的队列名"绑定上去了,所以路由键写队列名就能直达。

还有第四种 headers 交换机,按消息头键值对匹配,几乎没人用,知道存在即可,后面不再提。

### declare_exchange(名字, 类型, durable=True)

三个脚本唯一的差别就是这个 `type` 参数,它是路由行为的全部。其余参数:`durable=True` 交换机定义落盘,生产恒开;`auto_delete` 绑定全没了自动删,少用;`internal` 只许 broker 内部转发,基本不用。声明幂等、参数必须和已存在的一致(406),规矩和队列一样。

### bind:队列向交换机登记

注意绑定的方向,永远是队列绑到交换机,生产者不知道也不该知道有哪些队列。一个队列可以绑多个绑定键(一次登记多种口味),一个交换机可以挂任意多队列。

> **敲黑板:** 新增一个订阅方(比如数据分析想听听订单事件),自己声明队列、自己绑定就行,订单服务一行代码都不用改——这就是 MQ 三大价值里的"解耦"。

### drain 里的 get()——拉模式

前两讲消费都是订阅(iterator),broker 主动推。`queue.get()`(fanout.py 的 ⑤)是反过来主动拉一条:队列空就返回 None(fail=False 时;默认 fail=True 会抛 QueueEmpty 异常)。轮询、巡检、"脚本结尾把结果倒出来看看"这类一次性场景用它;常驻消费者永远用订阅,拉的实时性差,空轮询还浪费。三个脚本都要倒数据,`drain` 小函数每个带一份。

## 跑起来

三个脚本各跑各的,谁也不依赖谁:

```
uv run python "rabbitmq教程/03-三种交换机/fanout.py"
uv run python "rabbitmq教程/03-三种交换机/direct.py"
uv run python "rabbitmq教程/03-三种交换机/topic.py"
```

值得留意的一处:脚本跑完会把演示用的队列和交换机删掉(`queue.delete()` / `exchange.delete()`),所以你跑到管理台 Queues/Exchanges 页可能扑个空。想边跑边看,在 `drain` 之前加一句 `await asyncio.sleep(60)` 再跑,就有整整一分钟去管理台围观。

## 企业里怎么选

- 事件通知、多方订阅、未来可能加订阅方 → **topic**,给它留足扩展性,路由键用 `业务域.动作` 分好层;
- 一对一指派、固定通道(短信走短信、邮件走邮件)→ **direct**;
- 无差别广播(配置刷新、缓存失效)→ **fanout**。

实际项目里 topic 一家独大,因为它能表达另外两种:绑 `#` 就是 fanout,绑精确键就是 direct。很多团队的共识是:**拿不准就用 topic**,给路由键留出层次,以后怎么变都有余地。
