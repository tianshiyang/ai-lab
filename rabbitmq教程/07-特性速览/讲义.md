## 一、优先级队列(priority.py)

需求很朴素:客服工单里 VIP 的要插队先处理。RabbitMQ 支持,但又是"两头都设置才生效"的老规矩:

```python
queue = await channel.declare_queue(
    "ch7.priority",
    durable=True,
    arguments={"x-max-priority": 10},  # ① 队列这头:开"允许插队"的开关
)
message = aio_pika.Message(body, priority=9)  # ② 消息这头:带上优先级,0~10,大的先走
```

`priority.py` 全文:

```python
"""第 7 讲·优先级队列:加急的消息先被消费。

uv run python "rabbitmq教程/07-特性速览/priority.py"
"""

import asyncio
import os
from pathlib import Path

import aio_pika
from dotenv import load_dotenv

load_dotenv(Path(__file__).parents[2] / ".env")
RABBITMQ_URL = os.environ["RABBITMQ_URL"]


async def main() -> None:
    connection = await aio_pika.connect(RABBITMQ_URL)
    async with connection:
        channel = await connection.channel()

        queue = await channel.declare_queue(
            "ch7.priority",
            durable=True,
            arguments={"x-max-priority": 10},  # 队列这头:开"允许插队"的开关,一般给 10
        )
        await queue.purge()

        # 队列空着、没有消费者时,按 普通普通加急普通加急 的顺序发 5 条
        for priority, text in [
            (None, "普通工单-1"),
            (None, "普通工单-2"),
            (9, "加急工单-3(客户在催)"),
            (None, "普通工单-4"),
            (9, "加急工单-5(老板在催)"),
        ]:
            await channel.default_exchange.publish(
                aio_pika.Message(text.encode("utf-8"), priority=priority),  # 消息这头:带上优先级
                routing_key=queue.name,
            )
        print("按 [普通, 普通, 加急, 普通, 加急] 的顺序发了 5 条,现在按队列顺序取出来:")
        print("(priority 不带时默认 0,数值大的排前面)\n")

        order = []
        while True:
            message = await queue.get(no_ack=True, fail=False)
            if message is None:
                break
            order.append(message.body.decode("utf-8"))
        for i, text in enumerate(order, 1):
            print(f"  第 {i} 个被消费: {text}")

        print("\n两条加急的插到了最前面——但注意,插队发生在 broker 排队这一层;")
        print("消息已经被 prefetch 推到消费者手里囤着的话,就管不到了(见讲义)。")

        await queue.delete()


asyncio.run(main())
```

两头缺一不可:队列没开 `x-max-priority`,消息带 priority 也被无视(纯先进先出);队列开了、消息没带,大家都是默认 0,还是先进先出。脚本在没有消费者的前提下按 [普通,普通,加急,普通,加急] 发 5 条,取出来的顺序:

```
第 1 个被消费: 加急工单-3(客户在催)
第 2 个被消费: 加急工单-5(老板在催)
第 3 个被消费: 普通工单-1
...
```

两条加急的插到了最前面。两个泼冷水的点:

- **插队只发生在 broker 排队层。**消息一旦被 prefetch 推进消费者手里囤着,顺序就定死了——第 2 讲实验二里"李四囤了一摞重活"的同款问题。想保优先级,高优先级通道的 prefetch 得压小;
- **生产里用得少。**插队意味着 broker 要维护有序结构,有开销;而且"VIP 优先"这种需求往往有更干净的解法:VIP 单独一个队列、单独一组消费者,物理隔离比逻辑插队更可控。所以它是"知道有,慎用"的典型。

## 二、RPC 模式(rpc_server.py + rpc_client.py)

MQ 的本职是异步,但协议里留了一套做"请求-响应"的零件:消息属性里的 **reply_to**(回信地址)和 **correlation_id**(对暗号)。整个来回长这样:

```
  客户端                                        服务端
    │ ① 声明专属临时回信队列(连接一断自动删)         │
    │ ② 发请求:reply_to=回信队列名,                │
    │           correlation_id=本次暗号   ──────►  │ ③ 收到请求,算平方
    │                                           │ ④ 回信:发往 reply_to,
    │ ⑤ 回信队列收到消息,对上暗号才收   ◄──────────  │    暗号原样带回
```

两个终端配合跑:

```
终端 1:  uv run python "rabbitmq教程/07-特性速览/rpc_server.py"
终端 2:  uv run python "rabbitmq教程/07-特性速览/rpc_client.py"
```

`rpc_client.py` 全文(关键动作都带标号):

```python
"""第 7 讲·RPC 客户端:发"求平方"请求,守在自己专属的回复队列上等结果。

要问的数字写死在 NUMBERS 里,想多问几个就往里加——回信队列自始至终只有一个。

uv run python "rabbitmq教程/07-特性速览/rpc_client.py"
"""

import asyncio
import os
import uuid
from pathlib import Path

import aio_pika
from dotenv import load_dotenv

load_dotenv(Path(__file__).parents[2] / ".env")
RABBITMQ_URL = os.environ["RABBITMQ_URL"]

# 要问的数字:想多问几个就往里加,下面会挨个调一次 ask_square
NUMBERS = [7, 12]

# 暗号 → 等回信的 future:回信到达时按暗号找到是哪次提问在等
PENDING: dict[str, asyncio.Future[str]] = {}


async def ask_square(channel: aio_pika.Channel, callback_queue: aio_pika.Queue, n: int) -> str:
    """发一次"n 的平方是几"的请求,等到自己的那封回信才返回。"""
    correlation_id = str(uuid.uuid4())  # 本次提问的暗号
    future = PENDING[correlation_id] = asyncio.get_running_loop().create_future()

    # ② 请求消息带着"回信地址"(reply_to)和"暗号"(correlation_id)
    await channel.default_exchange.publish(
        aio_pika.Message(
            str(n).encode(), correlation_id=correlation_id, reply_to=callback_queue.name
        ),
        routing_key="ch7.rpc",
    )
    print(f"已发出请求: {n} 的平方是几?等待回信……")

    answer = await asyncio.wait_for(future, timeout=10)  # ③ 蹲在回信队列上等结果
    print(f"回信到了: {answer}")
    return answer


async def main() -> None:
    connection = await aio_pika.connect(RABBITMQ_URL)
    async with connection:
        channel = await connection.channel()

        # ① 不带名字 + exclusive:只属于本次连接的临时回信队列,声明一次问几次都用它
        callback_queue = await channel.declare_queue(exclusive=True)

        # 回信全进这一个队列,按暗号分给对应的提问;对不上暗号的(陈年旧回复)只确认不采信
        async def on_reply(message: aio_pika.IncomingMessage) -> None:
            future = PENDING.pop(message.correlation_id, None)
            if future is not None and not future.done():
                future.set_result(message.body.decode())
            await message.ack()

        await callback_queue.consume(on_reply)

        # 想问几个数就把 ask_square 多调几次,数字全写死在 NUMBERS 里
        for n in NUMBERS:
            await ask_square(channel, callback_queue, n)


asyncio.run(main())
```

`rpc_server.py` 全文:

```python
"""第 7 讲·RPC 服务端:收"求平方"的请求,算完把结果发回请求方留下的回复队列。

uv run python "rabbitmq教程/07-特性速览/rpc_server.py"     # 常驻,和 rpc_client.py 配合
"""

import asyncio
import os
from pathlib import Path

import aio_pika
from dotenv import load_dotenv

load_dotenv(Path(__file__).parents[2] / ".env")
RABBITMQ_URL = os.environ["RABBITMQ_URL"]


async def main() -> None:
    connection = await aio_pika.connect_robust(RABBITMQ_URL)
    async with connection:
        channel = await connection.channel()
        queue = await channel.declare_queue("ch7.rpc", durable=True)

        print("RPC 服务端已启动:发我一个数字字符串,我把它的平方算给你")

        async for message in queue.iterator():
            n = int(message.body.decode("utf-8"))
            print(f"  收到请求: {n} 的平方是几?  (correlation_id={message.correlation_id})")
            await asyncio.sleep(1)  # 假装是耗时计算

            # ① 结果发回请求方消息里指定的 reply_to 队列,原样带回 correlation_id 对暗号
            await channel.default_exchange.publish(
                aio_pika.Message(
                    str(n * n).encode("utf-8"),
                    correlation_id=message.correlation_id,
                ),
                routing_key=message.reply_to,
            )
            await message.ack()
            print(f"  已回复: {n * n}")


asyncio.run(main())
```

客户端把"问一次"封装成 `ask_square()`,要问的数字写死在 `NUMBERS` 里,想问几个就把函数多调几次;① 回信队列整个进程只声明一次(`declare_queue(exclusive=True)`,不带名字,连接一断自动删),② 每次提问都在消息里留下回信地址和暗号,③ 蹲在同一个回信队列上等自己的那封回信;服务端 ① 把结果发往 `message.reply_to`、暗号 `correlation_id` 原样带回。

为什么客户端要检查 correlation_id?回信队列是自己的没错,但几次提问共用这一个队列,回信混在一起,还可能有"陈年旧回复"(上一次超时了没等到,这回又来问),全靠暗号甄别哪封回信属于哪次提问。

> **敲黑板:** 面试爱问 RPC 模式(AMQP 的经典应用),但实际项目里同步调用都有更顺手的工具(HTTP、gRPC),MQ 硬做 RPC 是扬短避长——它的强项是异步解耦。见过、能讲出 reply_to/correlation_id 各干什么,够了。

## 三、quorum 队列(quorum.py)

前面所有队列都是 classic 类型(经典单机队列)。RabbitMQ 3.8 之后主推 **quorum 队列**(仲裁队列):基于 Raft 共识的多副本,消息在多数节点确认后才算写入,broker 挂一台,消息不丢不中断。声明的差别只有一行:

```python
queue = await channel.declare_queue(
    "ch7.quorum.demo",
    durable=True,  # quorum 必须 durable
    arguments={"x-queue-type": "quorum"},  # ① classic 换成 quorum,就这一个参数
)
```

`quorum.py` 全文:

```python
"""第 7 讲·quorum 队列:声明的差别就一个参数,跑一遍确认它存在。

uv run python "rabbitmq教程/07-特性速览/quorum.py"
"""

import asyncio
import os
from pathlib import Path

import aio_pika
from dotenv import load_dotenv

load_dotenv(Path(__file__).parents[2] / ".env")
RABBITMQ_URL = os.environ["RABBITMQ_URL"]


async def main() -> None:
    connection = await aio_pika.connect(RABBITMQ_URL)
    async with connection:
        channel = await connection.channel()

        queue = await channel.declare_queue(
            "ch7.quorum.demo",
            durable=True,  # quorum 队列必须 durable
            arguments={"x-queue-type": "quorum"},  # 就这一个参数,classic 换成 quorum
        )
        await channel.default_exchange.publish(
            aio_pika.Message("存进 quorum 队列的一条消息".encode()),
            routing_key=queue.name,
        )
        message = await queue.get(no_ack=True)
        print(f"声明成功,消息进出正常: {message.body.decode('utf-8')}")
        print("去管理台 Queues 页看 ch7.quorum.demo,Type 列是 quorum(普通队列是 classic)")

        await queue.purge()
        # quorum 队列不支持 delete 带 if_unused / if_empty 旗标(aio-pika 默认都开着),全关掉
        await queue.delete(if_unused=False, if_empty=False)


asyncio.run(main())
```

跑一遍,去管理台看 Type 列的区别(classic / quorum)。选型上记三句话:

- 单机玩玩,classic 足够;
- 生产集群 + 消息不能丢(订单、支付),官方建议新项目直接 quorum——它对第 4 讲"持久化仍可能丢最后零点几秒"的极端场景兜底更狠;
- 曾经还有个 lazy 队列(消息尽量落盘、内存只留索引,专为百万级堆积设计),3.12 以后 classic 队列自动具备了类似行为,lazy 单独提得越来越少。

> **敲黑板:** 换队列类型,客户端行为有细微差异。踩坑实录:quorum 队列的 delete 不支持 if_unused / if_empty 旗标,而 aio-pika 的 `delete()` 默认俩全开,得显式 `delete(if_unused=False, if_empty=False)`。这是"了解即可"类知识的典型代价:不踩不知道,踩了记得回来翻。

## 四、管理 HTTP API

管理台网页背后的接口全开放,认证后能做页面能做的一切。巡检积压、定时清理、自动化运维的标配:

```
# 所有队列:名字、消息数、消费者数

# 某个队列的详情
curl -s -u tianshiyang:tianshiyang http://62.234.16.180:15672/api/queues/lab/ch2.tasks
```

返回是 JSON,`jq` 过滤一下就是现成的监控数据源。生产上 Prometheus 抓 RabbitMQ 指标,底层就是这套。不必深学,会用 curl 查队列,排障时就快人一步。

## 五、一句话带过的名单

- **headers 交换机**:按消息头键值对匹配,和 direct 干一样的活但更啰嗦,几乎无人使用;
- **x-single-active-consumer**:一个队列多个消费者,但同一时刻只让一个收(一主多备),消费逻辑不支持并发时用;
- **集群、Federation、Shovel**:多节点高可用、跨机房联邦、消息搬运,运维侧的活,开发知道名字和用途即可;
- **流控与内存/磁盘告警**:broker 内存超线会 TCP 背压挡住发布方,管理台 Alerts 页能看到,排查"发布突然变慢"时想到它。
