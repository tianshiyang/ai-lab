# 第 5 讲 · Redis 分布式锁

## 0 模型

数据库锁管的是"数据库里的行"。但有一类互斥,锁的东西根本不在数据库里:

| 场景 | 要互斥的是什么 | 为什么数据库锁不够用 |
|---|---|---|
| 定时任务多实例部署 | "每晚 2 点的对账只能跑一份" | 任务跑 10 分钟,锁要跟着进程走,不是跟着某一行数据走 |
| 防重复触发 | "同一用户的导出任务,前一个没跑完不许再点" | 想拦在"开始干活之前",还不想为此占用数据库连接 |
| 缓存重建防击穿 | "缓存失效瞬间,只许一个请求去查库回填" | 加锁是为了**少打**数据库,再拿数据库行锁本末倒置 |

服务部署了 3 个实例,进程内的 `threading.Lock` 各自为政,门形同虚设——**互斥要放到所有进程都看得见的地方:Redis**。

```
   实例A ──SET lock:job token NX PX 30000──► ┐
   实例B ──SET lock:job token NX PX 30000──► Redis │ 谁先 SET 成功谁持锁
   实例C ──SET lock:job token NX PX 30000──► ┘     后来者 SET 失败(或等/或跳过)
        持锁者干活 → 干完用 Lua 校验 token 再删
```

## 1 正确的加锁:一条 SET 顶三条命令

```python
client.set(KEY, token, nx=True, px=30000)
```

四个要素,缺一个都是错的:

| 要素 | 作用 | 缺了会怎样 |
|---|---|---|
| `nx=True`(不存在才写) | 互斥的根本 | 变成普通 SET,人人都能"加锁成功" |
| `px=30000`(毫秒过期) | 持锁者暴毙后锁自动放 | 进程崩了不释放,锁变永久,业务死锁 |
| `token`(每次加锁的唯一随机串) | 证明"这锁是我的" | 释放时可能删掉**别人的锁**(本讲第二幕事故) |
| Lua 脚本释放(比对 token 再删) | 比对和删除原子 | "比对完、删之前"锁易主,照样误删 |

**敲黑板:老写法 `SETNX` 一条 + `EXPIRE` 一条是错的。** 两条命令之间进程崩了,锁永不过期。`SET key value NX PX 30000` 是一条命令,Redis 单线程执行,天然原子——这就是"加锁正确姿势"的全部秘密。

## 2 敲代码:`redis_lock_naive.py`(三幕含一场事故)

```python
"""三幕:正常加解锁 → 锁过期+裸DEL连环事故 → Lua安全释放止住误删。"""

import asyncio
import os
import time
from pathlib import Path
from uuid import uuid4

import redis
from dotenv import load_dotenv

load_dotenv(Path(__file__).parents[2] / ".env")
REDIS_URL = os.environ["REDIS_URL"]
KEY = "lock05:job"

# 释放锁的 Lua 脚本:是我加的锁才删,不是就不动——判断和删除必须一气呵成
RELEASE_LUA = """
if redis.call("get", KEYS[1]) == ARGV[1] then
    return redis.call("del", KEYS[1])
else
    return 0
end
"""


def log(message: str) -> None:
    print(f"[{time.strftime('%H:%M:%S')}] {message}", flush=True)


def acquire(client: redis.Redis, token: str, px: int) -> bool:
    """加锁:不存在才能写入,顺手带过期——一条命令,天然原子。"""
    return bool(client.set(KEY, token, nx=True, px=px))


def release_safe(client: redis.Redis, token: str) -> bool:
    """安全释放:token 对得上才删(Lua 保证判断+删除原子)。"""
    return bool(client.eval(RELEASE_LUA, 1, KEY, token))


async def worker(who: str, work_seconds: float, px: int, safe: bool) -> None:
    """抢锁干活:px 是锁的自动过期毫秒数,业务比锁先完成就没事。"""
    client = redis.Redis.from_url(REDIS_URL, decode_responses=True)
    token = uuid4().hex
    if not acquire(client, token, px):
        log(f"{who} 没抢到锁,走了")
        return
    log(
        f"{who} 抢到锁(token={token[:6]}…),干活 {work_seconds} 秒(锁 {px / 1000} 秒自动过期)"
    )
    await asyncio.sleep(work_seconds)
    if safe:
        mine = release_safe(client, token)
        log(
            f"{who} 干完,{'锁还给了自己删掉' if mine else '锁早已不是自己的,Lua 拒绝删'}"
        )
    else:
        client.delete(KEY)
        log(f"{who} 干完,裸 DEL 删锁——不管锁是谁的都删")


async def prober(who: str, at: float, hold: float = 0.8) -> None:
    """定时探针:到点试一次加锁,拿得到说明此刻没人在临界区里。"""
    await asyncio.sleep(at)
    client = redis.Redis.from_url(REDIS_URL, decode_responses=True)
    token = uuid4().hex
    if not acquire(client, token, 6000):
        log(f"{who} 在 {at} 秒处试锁:进不去,里面有人")
        return
    log(f"{who} 在 {at} 秒处试锁:!!! 进来了 !!!")
    await asyncio.sleep(hold)
    release_safe(client, token)
    log(f"{who} 退出去(删的是自己的锁)")


async def main() -> None:
    client = redis.Redis.from_url(REDIS_URL, decode_responses=True)

    log("=== 第一幕:锁 5 秒过期,活 1.5 秒干完,一切正常 ===")
    client.delete(KEY)
    await asyncio.gather(
        worker("甲", 1.5, 5000, safe=True), prober("乙", 0.5), prober("丙", 2.2)
    )

    log("=== 第二幕:锁 1.5 秒就过期,活 3 秒,还用裸 DEL ===")
    client.delete(KEY)
    await asyncio.gather(
        worker("甲", 3.0, 1500, safe=False), prober("乙", 2.0, hold=3.0)
    )
    log(
        "复盘:锁 1.5 秒过期→乙 2 秒进来(甲还没干完,已经两人重叠)→甲 3 秒裸 DEL→乙的锁没了"
    )
    await asyncio.sleep(0.3)

    log("=== 第三幕:同样超时,但甲改用 Lua 安全释放 ===")
    client.delete(KEY)
    await asyncio.gather(
        worker("甲", 3.0, 1500, safe=True), prober("乙", 2.0, hold=3.0)
    )
    log("复盘:Lua 认 token,甲删不掉乙的锁;但甲乙重叠(锁过期)依然在——这要靠下一节的续期")
    client.delete(KEY)


asyncio.run(main())
```

第二幕的连环事故逐帧回放(这就是生产上"分布式锁偶尔失灵"的解剖图):

```
t=0.0  甲 SET NX 成功,锁 1.5 秒过期
```

**敲黑板:过期时间是双刃剑——没有它,持锁者崩溃就是全局死锁;有它,业务超过时长就是互斥失效。** 两条出路:①估算业务最长耗时,过期时间放宽 3~5 倍;②看门狗续期(下一节)。

## 4 看门狗:锁跟着人活着

思路:过期时间照设(比如 2 秒),但**持锁期间后台有个小任务每 0.6 秒把过期时间续回 2 秒**。人活着,锁就活着;人暴毙,续期随之消失,锁 2 秒后自动放。Java 的 Redisson 内置了这套;Python 的 redis-py 只提供 `refresh()`,续期循环自己写(几行):

```python
"""redis_lock_watchdog.py —— 完整代码见下,核心是 watchdog 协程"""
```

```python
"""看门狗续期:活没干完锁不过期;进程崩了没人续期,锁自动放。"""

import asyncio
import os
import time
from pathlib import Path
from uuid import uuid4

import redis
from dotenv import load_dotenv

load_dotenv(Path(__file__).parents[2] / ".env")
REDIS_URL = os.environ["REDIS_URL"]
KEY = "lock05:watchdog"

# 续期脚本:还是我的锁就把过期时间往后推
RENEW_LUA = """
if redis.call("get", KEYS[1]) == ARGV[1] then
    return redis.call("pexpire", KEYS[1], ARGV[2])
else
    return 0
end
"""

RELEASE_LUA = """
if redis.call("get", KEYS[1]) == ARGV[1] then
    return redis.call("del", KEYS[1])
else
    return 0
end
"""


def log(message: str) -> None:
    print(f"[{time.strftime('%H:%M:%S')}] {message}", flush=True)


def acquire(client: redis.Redis, token: str, px: int) -> bool:
    return bool(client.set(KEY, token, nx=True, px=px))


async def watchdog(client: redis.Redis, token: str, px: int) -> None:
    """看门狗:每隔 px/3 毫秒续一次期,只要进程活着,锁就一直活着。"""
    while True:
        await asyncio.sleep(px / 1000 / 3)
        ok = bool(client.eval(RENEW_LUA, 1, KEY, token, px))
        if not ok:
            log("看门狗:锁没了(被抢/被删),停止续期")
            return


async def worker_with_dog(work_seconds: float, crash: bool) -> None:
    """带看门狗干活:crash=True 模拟进程暴毙(不释放、看门狗一起死)。"""
    client = redis.Redis.from_url(REDIS_URL, decode_responses=True)
    token = uuid4().hex
    if not acquire(client, token, 2000):
        log("worker 没抢到锁")
        return
    log(f"worker 抢到锁,干活 {work_seconds} 秒,看门狗已上岗(每 0.67 秒续到 2 秒)")
    dog = asyncio.create_task(watchdog(client, token, 2000))
    await asyncio.sleep(work_seconds)
    if crash:
        dog.cancel()  # 进程崩了,续期任务和人一起没了
        log("worker 暴毙!不释放锁,看门狗也死了——就当这个进程不存在")
        return
    dog.cancel()
    released = bool(client.eval(RELEASE_LUA, 1, KEY, token))
    log(f"worker 干完正常释放:{'成功' if released else '失败'}")


async def prober(who: str, at: float) -> None:
    await asyncio.sleep(at)
    client = redis.Redis.from_url(REDIS_URL, decode_responses=True)
    ok = bool(client.set(KEY, uuid4().hex, nx=True, px=1000))
    if ok:
        client.delete(KEY)
    log(f"{who} 在 {at} 秒处试锁:{'进来了(锁是空的)' if ok else '进不去(锁还活着)'}")


async def main() -> None:
    client = redis.Redis.from_url(REDIS_URL, decode_responses=True)

    log("=== 第一幕:活干 5 秒,锁只有 2 秒,全靠看门狗续命 ===")
    client.delete(KEY)
    await asyncio.gather(
        worker_with_dog(5.0, crash=False),
        prober("探针-1秒", 1),
        prober("探针-3秒", 3),
        prober("探针-4.5秒", 4.5),
        prober("探针-6秒", 6),
    )

    log("=== 第二幕:进程暴毙,没人续期,锁 2 秒后自动放 ===")
    client.delete(KEY)
    await asyncio.gather(
        worker_with_dog(1.0, crash=True),
        prober("探针-0.5秒", 0.5),
        prober("探针-3秒", 3),
    )
    client.delete(KEY)


asyncio.run(main())
```

## 5 实战:直接用 redis-py 自带的 Lock

上面手写是为了看穿原理。真写业务,redis-py 的 `client.lock()` 把 token、Lua 释放全包了,惯用形状是:

```python
lock = client.lock(KEY, timeout=5, blocking=False)  # timeout=过期秒数
if not lock.acquire():
    ...  # 抢不到,fast-fail
try:
    ...  # 临界区
    lock.refresh()  # 干久了续一下(注意:不会自动续,得自己调)
finally:
    lock.release()  # 内部就是"比对 token 再删"的 Lua
```

`redis_lock_builtin.py` 全文——注意 `try/finally` 的形状,出了任何事锁都必须还:

```python
"""redis-py 自带的 Lock:token、Lua 释放全内置,直接用就行。"""

import asyncio
import os
import time
from pathlib import Path

import redis
from dotenv import load_dotenv

load_dotenv(Path(__file__).parents[2] / ".env")
REDIS_URL = os.environ["REDIS_URL"]
KEY = "lock05:builtin"


def log(message: str) -> None:
    print(f"[{time.strftime('%H:%M:%S')}] {message}", flush=True)


async def risky_job(who: str) -> None:
    client = redis.Redis.from_url(REDIS_URL, decode_responses=True)
    lock = client.lock(KEY, timeout=5, blocking=False)  # blocking=False:抢不到立刻走
    if not lock.acquire():
        log(f"{who} 抢不到锁,fast-fail")
        return
    try:
        log(f"{who} 拿到锁,干活(这次中途会出事)")
        await asyncio.sleep(1)
        raise RuntimeError("下游接口炸了")
    except RuntimeError as e:
        log(f"{who} 出事了:{e}")
    finally:
        lock.release()  # 内置 Lua 按 token 安全释放
        log(f"{who} finally 里把锁还了——出了事也不许占着")


async def next_job(who: str) -> None:
    await asyncio.sleep(2)
    client = redis.Redis.from_url(REDIS_URL, decode_responses=True)
    lock = client.lock(KEY, timeout=5, blocking=False)
    if lock.acquire():
        try:
            log(f"{who} 拿到锁(前人即使出事也把锁还干净了)")
            lock.release()
        finally:
            pass


async def main() -> None:
    await asyncio.gather(risky_job("甲"), next_job("乙"))


asyncio.run(main())
```

```
[11:06:21] 乙 拿到锁(前人即使出事也把锁还干净了)
```

| 参数 | 含义 | 备注 |
|---|---|---|
| `timeout` | 过期秒数 | 不设则永不过期——**除非看门狗在续,否则一定要设** |
| `blocking=False` | 抢不到立刻返回 False | 协程里推荐 fast-fail + 自己 `asyncio.sleep` 重试;`blocking=True` 内部是小睡轮询,会卡事件循环 |
| `blocking_timeout` | 最多等多久 | 搭配 blocking=True 用 |
| `lock.refresh()` | 续期 | 手动调,没有自动看门狗 |

## 6 边界与争议:分布式锁不是银弹

| 问题 | 结论 |
|---|---|
| 主从切换丢锁 | Redis 异步复制:主节点写入锁后没来得及同步就挂,从顶上,锁没了。**单实例 Redis 或等待充分复制的部署都回避不了本质:Redis 的锁是"尽力互斥"** |
| Redlock | 向多个独立 Redis 节点同时加锁、过半成功才算持有。antirez(Redis 作者)推荐,Martin Kleppmann(《DDIA》作者)发文质疑其依赖时钟假设。企业实际:**用得不多** |
| 严格互斥怎么办 | 上数据库行锁(能容忍慢)、ZooKeeper/etcd 临时节点(分布式协调,强一致)。支付清算级别的互斥别用 Redis 锁 |
| Redis Cluster | 单 key 的 SET/Lua 完全不受影响(锁就是单 key);多 key 操作才要操心 hash tag。工作的 Redis Cluster 用 `get_redis_cluster()` 拿客户端,锁用法不变 |
| 锁粒度 | 锁"业务对象"(如 `lock:export:{user_id}`),别锁全局,否则一个用户卡住所有人 |
| 锁内代码 | 仍要幂等、仍要短——锁只是降低并发,不是事务 |

**敲黑板:选型顺序——能不分布式锁就不分布式锁。** 单机内存锁(异步教程第 4 讲的 `threading.Lock` / `asyncio.Lock`)能解决的别上 Redis;数据库能管的(改行、防重)用第 2~4 讲的数据库锁;剩下的"跨进程、非严格"场景,才是 Redis 锁的主场。
